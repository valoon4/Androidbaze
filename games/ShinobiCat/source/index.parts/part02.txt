function clamp(v, min, max) {
  return Math.max(min, Math.min(max, v));
}

function circleHit(a, b, extra = 0) {
  return Math.hypot(a.x - b.x, a.y - b.y) < a.r + b.r + extra;
}

function circleRectHit(x, y, r, rect) {
  const nearestX = clamp(x, rect.x, rect.x + rect.w);
  const nearestY = clamp(y, rect.y, rect.y + rect.h);
  return Math.hypot(x - nearestX, y - nearestY) < r;
}

function resolveCircleObstacle(entity, tx, ty, obstacle) {
  const min = entity.r + obstacle.r;
  let x = tx, y = ty;
  if (Math.hypot(x - obstacle.x, entity.y - obstacle.y) < min) x = entity.x;
  if (Math.hypot(x - obstacle.x, y - obstacle.y) < min) y = entity.y;
  const rx = x - obstacle.x, ry = y - obstacle.y;
  const d = Math.hypot(rx, ry) || 1;
  if (d < min) { x = obstacle.x + rx / d * min; y = obstacle.y + ry / d * min; }
  return {x, y};
}

function movePlayer(p, mx, my) {
  if (!mx && !my) return;
  let tx = clamp(p.x + mx, 38, 922);
  let ty = clamp(p.y + my, 62, 500);

  if (G.area === 'secretBow') {
    if (circleRectHit(tx, p.y, p.r, ARROW_CHASM)) tx = p.x;
    if (circleRectHit(tx, ty, p.r, ARROW_CHASM)) ty = p.y;
  }
  if (G.area === 'mimic') {
    const resolved = resolveCircleObstacle(p, tx, ty, MIMIC_STONE);
    tx = resolved.x; ty = resolved.y;
  }
  if(G.area==='astralGauntlet'||G.area==='astralBoss'){
    for(const obstacle of ASTRAL_PILLARS){const resolved=resolveCircleObstacle(p,tx,ty,obstacle);tx=resolved.x;ty=resolved.y;}
  }

  const b = G.block;
  if (!b) {
    p.x = tx;
    p.y = ty;
    return;
  }

  const min = p.r + b.r;
  if (Math.hypot(tx - b.x, ty - b.y) >= min) {
    p.x = tx;
    p.y = ty;
    return;
  }

  const inputX = tx - p.x;
  const inputY = ty - p.y;
  const relX = b.x - p.x;
  const relY = b.y - p.y;
  const pushHorizontal = Math.abs(relX) >= Math.abs(relY);

  if (!G.flags.p2) {
    if (pushHorizontal && Math.abs(relY) < 34 && Math.sign(inputX) === Math.sign(relX) && Math.abs(inputX) > Math.abs(inputY) * 0.65) {
      const step = inputX;
      const next = b.x + step;
      if (next > 112 && next < 848) {
        b.x = next;
        p.x = b.x - Math.sign(relX) * min;
        p.y += (b.y - p.y) * 0.10;
        return;
      }
    }
    if (!pushHorizontal && Math.abs(relX) < 34 && Math.sign(inputY) === Math.sign(relY) && Math.abs(inputY) > Math.abs(inputX) * 0.65) {
      const step = inputY;
      const next = b.y + step;
      if (next > 105 && next < 438) {
        b.y = next;
        p.y = b.y - Math.sign(relY) * min;
        p.x += (b.x - p.x) * 0.10;
        return;
      }
    }
  }

  if (Math.hypot(tx - b.x, p.y - b.y) >= min) p.x = tx;
  if (Math.hypot(p.x - b.x, ty - b.y) >= min) p.y = ty;
  const rx = p.x - b.x;
  const ry = p.y - b.y;
  const d = Math.hypot(rx, ry) || 1;
  if (d < min) {
    p.x = b.x + rx / d * min;
    p.y = b.y + ry / d * min;
  }
}

function hurtPlayer(baseDamage) {
  const p = G.player;
  if (p.invuln > 0 || p.respawnGrace > 0) return;
  const reduction = Math.max(0, currentDef() - 1);
  const damage = Math.max(1, baseDamage - reduction);
  p.hp = Math.max(0, p.hp - damage);
  p.invuln = 0.9;
  updateHud();

  if (p.hp <= 0) {
    const dungeonDeath = G.area !== 'forest';
    openMessage('K.O.', dungeonDeath ? 'Du wirst zum Eingang des Dungeons zurückgebracht.' : 'Du wachst auf der Waldlichtung auf.', () => {
      p.hp = p.maxHp;
      p.invuln = 0;
      p.respawnGrace = 1.2;
      p.slow = 0;
      p.attackCd = 0;
      p.attackFx = 0;
      resetInput();
      if (G.area && G.area.startsWith('desert')) enterArea('desertEntry', DUNGEON_SPAWN);
      else if (G.area && G.area.startsWith('astral')) enterArea('astralEntry', DUNGEON_SPAWN);
      else if (dungeonDeath) enterArea('entry', DUNGEON_SPAWN);
      else enterArea('forest', {x: 480, y: 410});
      updateHud();
    });
  }
}

function nearestInteraction(maxDistance = 76) {
  if (!G || !['entry','desertEntry','astralEntry'].includes(G.area)) return null;
  const p=G.player;
  const targets=[
    {kind:'blue',d:Math.hypot(p.x-INTERACTIVES.blue.x,p.y-INTERACTIVES.blue.y)},
    {kind:'red',d:Math.hypot(p.x-INTERACTIVES.red.x,p.y-INTERACTIVES.red.y)},
    {kind:'wardrobe',d:Math.hypot(p.x-INTERACTIVES.wardrobe.x,p.y-INTERACTIVES.wardrobe.y)}
  ].sort((a,b)=>a.d-b.d);
  return targets[0].d<=maxDistance?targets[0]:null;
}

function weaponDown() {
  if (!G || screen !== 'game' || paused) return;
  const weapon = G.player.weapon;
  if (weapon === 'sword') swordAttack();
  else if (weapon === 'bow') shootArrow(false);
  else if (weapon === 'bomb') placeBomb(false);
  else if (weapon === 'mirror') activateMirror();
  else if (weapon === 'boomerang') {
    if (!G.weapons.boomerang || G.boomerangs.some(b => b.owner === 'player')) return;
    G.player.boomerangHold = true;
    G.player.holdTime = 0;
    G.player.lockTargets = [];
    G.player.lockArmed = true;
    G.player.lastLockFace = null;
  }
}

    releaseBoomerang(false);
  }

function activateMirror(){
  const p=G.player;

function swordAttack() {
  const p = G.player;
  if (p.attackCd > 0) return;
  p.attackCd = 0.34;
  p.attackFx = 0.24;
  const ax = p.x + p.face.x * 48;
  const ay = p.y + p.face.y * 48;

  if (G.area === 'forest') {
    for (const bush of G.bushes) {
      if (!bush.cut && Math.hypot(bush.x - ax, bush.y - ay) < 52) {
        bush.cut = true;
        bush.respawn = 12;
        if (Math.random() < 0.65) {
          spawnDrop(Math.random() < 0.78 ? 'coin' : 'heart', bush.x, bush.y, Math.random() < 0.15 ? 5 : 1);
        }
      }
    }
  }

  hitCombatTargets(ax, ay, 47, currentAtk(), 0);
  mimicCopies('sword');
}

function hitCombatTargets(x, y, range, damage, stun) {
  for (const e of G.enemies) {
    if (!e.dead && e.hp > 0 && Math.hypot(e.x - x, e.y - y) < e.r + range) {
      damageEnemy(e, damage, stun);
    }
  }
  if (G.boss && G.boss.hp > 0 && Math.hypot(G.boss.x - x, G.boss.y - y) < G.boss.r + range) {
    damageBoss(damage);
  }
  if (G.mimic && G.mimic.hp > 0 && Math.hypot(G.mimic.x - x, G.mimic.y - y) < G.mimic.r + range) {
    damageMimic(damage);
  }
}

function projectileMirrorDirection(projectile, fallback) {
  if (projectile) {
    const px = projectile.dx !== undefined ? projectile.dx : projectile.vx;
    const py = projectile.dy !== undefined ? projectile.dy : projectile.vy;
    const len = Math.hypot(px, py) || 1;
    return {x: -px / len, y: py / len};
  }
  return mirrorFace(fallback);
}

function shootArrow(fromMimic, copiedProjectile = null) {
  const p = G.player;
  if (!fromMimic) {
    if (!G.weapons.bow) return;
    if (G.ammo.arrows <= 0) return dryWeaponFeedback();
    if (p.attackCd > 0) return;
    G.ammo.arrows--;
    p.attackCd = 0.3;
  }
  const source = fromMimic ? G.mimic : p;
  if (!source) return;
  const face = fromMimic ? projectileMirrorDirection(copiedProjectile, p.face) : p.face;
  const speed = 430;
  const projectile = {
    x: source.x + face.x * 28, y: source.y + face.y * 28,
    dx: face.x * speed, dy: face.y * speed,
    r: 7, t: 0, owner: fromMimic ? 'enemy' : 'player',
    damage: fromMimic ? 1 : currentAtk(), type: 'arrow'
  };
  if (fromMimic) G.enemyProjectiles.push(projectile);
  else {
    G.arrows.push(projectile);
    mimicCopies('bow', projectile);
    updateWeaponButtons();
  }
  return projectile;
}

function placeBomb(fromMimic) {
  const p = G.player;
  if (!fromMimic) {
    if (!G.weapons.bomb) return;
    if (G.ammo.bombs <= 0) return dryWeaponFeedback();
    if (G.bombs.length > 0) return dryWeaponFeedback();
    G.ammo.bombs--;
  }
  const source = fromMimic ? G.mimic : p;
  if (!source) return;
  const bomb = {x: source.x, y: source.y, r: 14, timer: 3, owner: fromMimic ? 'enemy' : 'player'};
  if (fromMimic) G.dungeon.mimic.bombs.push(bomb);
  else {
    G.bombs.push(bomb);
    mimicCopies('bomb');
    updateWeaponButtons();
  }
}

function releaseBoomerang(fromMimic, copiedProjectile = null) {
  const p = G.player;
  if (!fromMimic) {
    p.boomerangHold = false;
    if (!G.weapons.boomerang || G.boomerangs.some(b => b.owner === 'player')) return;
  }
  const source = fromMimic ? G.mimic : p;
  if (!source) return;
  const face = fromMimic ? projectileMirrorDirection(copiedProjectile, p.face) : p.face;
  const targets = fromMimic ? [] : p.lockTargets.map(t => ({kind: t.kind, ref: t.ref}));
  const b = {
    x: source.x + face.x * 25, y: source.y + face.y * 25,
    vx: face.x * 420, vy: face.y * 420, r: 13,
    owner: fromMimic ? 'enemy' : 'player', phase: 'out', life: 0,
    hitIds: {}, targets, targetIndex: 0
  };
  G.boomerangs.push(b);
  if (!fromMimic) {
    p.lockTargets = [];
    p.lockArmed = true;
    p.lastLockFace = null;
    mimicCopies('boomerang', b);
  }
  return b;
}

function mirrorFace(face) {
  return {x: -face.x, y: face.y};
}

function mimicCopies(action, sourceProjectile = null) {
  if (!G.mimic || G.mimic.hp <= 0) return;
  if (action === 'sword') {
    G.mimic.attackFx = 0.24;
    const face = mirrorFace(G.player.face);
    const ax = G.mimic.x + face.x * 48;
    const ay = G.mimic.y + face.y * 48;
    if (Math.hypot(G.player.x - ax, G.player.y - ay) < G.player.r + 45) hurtPlayer(1);
  } else if (action === 'bow') {
    shootArrow(true, sourceProjectile);
  } else if (action === 'bomb') {
    if (G.dungeon.mimic.bombs.length === 0) placeBomb(true);
  } else if (action === 'boomerang') {
    // Every throw gets a fresh mirrored counterpart; an older enemy throw is recalled.
    for (const old of G.boomerangs) if (old.owner === 'enemy') old.dead = true;
    releaseBoomerang(true, sourceProjectile);
  }
}

function acquireBoomerangTargets() {
  const p = G.player;
  const stillValid = t => {
    const ref = t.ref;
    if (!ref || ref.dead || ref.hp === 0) return false;
    if (t.kind === 'switch' && ref.on) return false;
    return true;
  };
  p.lockTargets = p.lockTargets.filter(stillValid).slice(0, 3);
  if (p.lockTargets.length >= 3) return;

  if (!p.lockArmed) {
    const last = p.lastLockFace || p.face;
    const turnDot = last.x * p.face.x + last.y * p.face.y;
    if (turnDot < 0.76) p.lockArmed = true;
    else return;
  }

  const locked = new Set(p.lockTargets.map(t => t.id));
  const candidates = [];
  const add = (kind, ref, id) => {
    if (!ref || locked.has(id) || ref.dead || ref.hp === 0 || ref.on) return;
    const dx = ref.x - p.x, dy = ref.y - p.y;
    const dist = Math.hypot(dx, dy);
    if (dist > 560 || dist < 25) return;
    const dot = (dx / dist) * p.face.x + (dy / dist) * p.face.y;
    if (dot < 0.88) return;
    candidates.push({kind, ref, id, score: dist + (1 - dot) * 240});
  };
  for (const sw of G.switches) if (sw.kind === 'boomerang' && !sw.on) add('switch', sw, sw.id);
  for (const e of G.enemies) if (!e.dead && e.hp > 0) add('enemy', e, e.id);
  if (G.boss && G.boss.hp > 0) add('boss', G.boss, 'boss');
  if (G.mimic && G.mimic.hp > 0) add('mimic', G.mimic, 'mimic');
  candidates.sort((a, b) => a.score - b.score);
  if (!candidates.length) return;
  p.lockTargets.push(candidates[0]);
  p.lastLockFace = {x: p.face.x, y: p.face.y};
  p.lockArmed = false;
}

function damageEnemy(e, damage, stun = 0) {
  if (e.dead || e.hp <= 0) return;
  e.hp = Math.max(0, e.hp - damage);
  e.hit = 0.2;
  if (stun > 0) e.stun = Math.max(e.stun, stun);
  if (e.hp <= 0) {
    e.dead = true;
    rollEnemyDrop(e);
  }
}

function damageBoss(damage) {
  if (!G.boss || G.boss.hp <= 0) return;
  if(G.boss.kind==='eclipse'&&G.boss.stage===1)return;
  G.boss.hp = Math.max(0, G.boss.hp - damage);
}

function damageMimic(damage) {
  if (!G.mimic || G.mimic.hp <= 0) return;
  G.mimic.hp = Math.max(0, G.mimic.hp - damage);
  if (G.mimic.hp <= 0 && !G.flags.mimicDefeated) {
    G.flags.mimicDefeated = true;
    spawnDrop('heartContainer', G.mimic.x, G.mimic.y, 1, 'mimic');
  }
}

function rollEnemyDrop(e) {
  if (e.dropRolled) return;
  e.dropRolled = true;
  if (['zombie','scarab','mummy','wisp','starKnight'].includes(e.type)) {
    const zombieCoins = rollCoinValue();
    if (zombieCoins > 0) spawnDrop('coin', e.x, e.y, zombieCoins);
    return;
  }
  const chance = e.type === 'slime' ? 0.50 : 0.40;
  if (Math.random() < chance) {
    spawnDrop(e.type === 'slime' ? 'slimeMaterial' : 'batMaterial', e.x, e.y);
    return;
  }
  const coins = rollCoinValue();
  if (coins > 0) spawnDrop('coin', e.x, e.y, coins);
}

function rollCoinValue() {
  const r = Math.random();
  if (r < 0.25) return 0;
  if (r < 0.50) return 1;
  if (r < 0.70) return 2;
  if (r < 0.90) return 5;
  if (r < 0.98) return 10;
  return 100;
}

function spawnDrop(type, x, y, value = 1, source = '') {
  G.drops.push({type, x, y, r: type === 'heartContainer' ? 18 : 11, value, area: G.area, source, t: 0, dead: false});
}

function useItem(item) {
  if (G.inventory[item] <= 0) return;
  if (item === 'apple') {
    if (G.player.hp >= G.player.maxHp) return toast('Deine Herzen sind voll');
    G.inventory.apple--;
    G.player.hp = Math.min(G.player.maxHp, G.player.hp + 2);
  } else if (item === 'energy') {
    G.inventory.energy--;
    G.buffs.attack.push(30);
  } else if (item === 'wind') {
    G.inventory.wind--;
    G.buffs.speed.push(30);
  } else if (item === 'bark') {
    G.inventory.bark--;
    G.buffs.defense.push(30);
  }
  renderInventory();
  updateHud();
}
