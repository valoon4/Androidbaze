function midiHz(note){return 440*Math.pow(2,(note-69)/12);}
function initMusic(){
  if(MUSIC.ctx)return true;
  const root=typeof window!=='undefined'?window:(typeof globalThis!=='undefined'?globalThis:{});
  const AC=root.AudioContext||root.webkitAudioContext;
  if(!AC)return false;
  try{
    MUSIC.ctx=new AC();MUSIC.master=MUSIC.ctx.createGain();MUSIC.master.gain.value=.0001;MUSIC.master.connect(MUSIC.ctx.destination);
    if(typeof setInterval==='function')MUSIC.timer=setInterval(musicTick,70);
    return true;
  }catch(_){return false;}
}
function scheduleTone(note,time,duration,type,volume,detune=0){
  if(note==null||!MUSIC.ctx||!MUSIC.master)return;
  const o=MUSIC.ctx.createOscillator(),g=MUSIC.ctx.createGain();
  o.type=type;o.frequency.setValueAtTime(midiHz(note),time);o.detune.setValueAtTime(detune,time);
  g.gain.setValueAtTime(.0001,time);g.gain.exponentialRampToValueAtTime(Math.max(.0002,volume),time+.025);
  g.gain.exponentialRampToValueAtTime(.0001,time+duration);
  o.connect(g);g.connect(MUSIC.master);o.start(time);o.stop(time+duration+.04);
}
function scheduleMusicStep(track,step,time){
  const beat=60/track.bpm/2,lead=track.lead[step%track.lead.length],bass=track.bass[step%track.bass.length];
  scheduleTone(lead,time,beat*.82,track.wave,track.volume);
  if(lead!=null&&step%4===2)scheduleTone(lead+12,time+.015,beat*.42,'sine',track.volume*.22,5);
  scheduleTone(bass,time,beat*1.75,'triangle',track.volume*.72);
  if(step%4===0)scheduleTone(31,time,beat*.18,'sine',track.volume*.9);
}
function musicTick(){
  if(!MUSIC.ctx||!MUSIC.track||!MUSIC_TRACKS[MUSIC.track])return;
  const track=MUSIC_TRACKS[MUSIC.track],beat=60/track.bpm/2,until=MUSIC.ctx.currentTime+.24;
  if(MUSIC.next<MUSIC.ctx.currentTime-.2)MUSIC.next=MUSIC.ctx.currentTime+.03;
  while(MUSIC.next<until){scheduleMusicStep(track,MUSIC.step++,MUSIC.next);MUSIC.next+=beat;}
}
function startMusic(){
  if(!initMusic())return;
  if(MUSIC.ctx.resume)MUSIC.ctx.resume();
  MUSIC.master.gain.cancelScheduledValues(MUSIC.ctx.currentTime);
  MUSIC.master.gain.setTargetAtTime(.78,MUSIC.ctx.currentTime,.08);
}
function setMusicTrack(track){
  if(!MUSIC.ctx&&track)return;
  if(MUSIC.track===track)return;
  MUSIC.track=track;MUSIC.step=0;
  if(MUSIC.ctx)MUSIC.next=MUSIC.ctx.currentTime+.05;
  if(MUSIC.master&&MUSIC.ctx){
    MUSIC.master.gain.cancelScheduledValues(MUSIC.ctx.currentTime);
    MUSIC.master.gain.setTargetAtTime(track ? .78 : .0001,MUSIC.ctx.currentTime,.12);
  }
}

function ensureRoomEnemies(area, type, n) {
  if (G.roomEnemies[area]) return;
  const positions=[[250,205],[480,180],[710,210],[340,350],[620,350],[480,300]];
  const arr = [];
  const stats={slime:{r:24,hp:2},bat:{r:18,hp:2},zombie:{r:23,hp:3},scarab:{r:19,hp:2},mummy:{r:25,hp:4},wisp:{r:18,hp:2},starKnight:{r:25,hp:4}};
  const st=stats[type]||stats.slime;
  for (let i = 0; i < n; i++) {
    const pos=positions[i%positions.length];
    arr.push({
      id: `${area}-${i}`, type, x:pos[0], y:pos[1], r:st.r,
      hp:st.hp,maxHp:st.hp,t:Math.random()*6,hit:0,stun:0,
      attackCd:.5+Math.random()*1.4,lunge:0,side:i%2?1:-1,dead:false,dropRolled:false
    });
  }
  G.roomEnemies[area] = arr;
}

function currentAtk() {
  return G.player.atk + G.buffs.attack.length;
}

function currentDef() {
  return G.player.def + G.buffs.defense.length;
}

function currentSpeed() {
  return G.player.speed * (1 + G.buffs.speed.length * 0.25);
}

function heartMarkup() {
  let out = '';
  const count = Math.ceil(G.player.maxHp / 2);
  for (let i = 0; i < count; i++) {
    const amount = G.player.hp - i * 2;
    const cls = amount >= 2 ? 'full' : amount === 1 ? 'half' : 'empty';
    out += `<span class="heart ${cls}">♥</span>`;
  }
  return out;
}

function updateHud() {
  if (!G) return;
  ui.hearts.innerHTML = heartMarkup();
  const buffIcons = [];
  if (G.buffs.attack.length) buffIcons.push(`⚡${G.buffs.attack.length}`);
  if (G.buffs.speed.length) buffIcons.push(`🪽${G.buffs.speed.length}`);
  if (G.buffs.defense.length) buffIcons.push(`🛡${G.buffs.defense.length}`);
  ui.stats.innerHTML = `<span>🪙 ${G.coins}</span><span>ATK ${currentAtk()}</span><span>DEF ${currentDef()}</span>${buffIcons.length ? `<small>${buffIcons.join(' ')}</small>` : ''}`;

}
